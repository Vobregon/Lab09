           const datos = ["Manzana", "Pera", "sandia"];

           function render(items){
            contenedor.innerHTML = "";
            items.forEach(fruta => {
                const li =  document.createElement("li");
                li.className = "bg-white p-2 rounded shadow"
                li.textContent = fruta;
